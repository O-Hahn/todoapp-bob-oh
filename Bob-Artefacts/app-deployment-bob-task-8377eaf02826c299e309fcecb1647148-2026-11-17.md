export class MemoryUserRepository implements IUserRepository {
  private users = new Map<string, UserDoc>();

  async create(user: UserDoc): Promise<UserDoc> {
    if (this.users.has(user._id)) {
      const err = new Error('Document update conflict') as any;
      err.status = 409;
      err.statusCode = 409;
      throw err;
    }
    const doc: UserDoc = { ...user, _rev: '1-rev' };
    this.users.set(user._id, doc);
    return { ...doc };
  }

  async getById(id: string): Promise<UserDoc | null> {
    const doc = this.users.get(id);
    return doc ? { ...doc } : null;
  }

  async update(user: UserDoc): Promise<UserDoc> {
    const existing = this.users.get(user._id);
    if (!existing) {
      const err = new Error('Document not found') as any;
      err.status = 404;
      err.statusCode = 404;
      throw err;
    }
    const doc: UserDoc = { ...user, _rev: '2-rev' };
    this.users.set(user._id, doc);
    return { ...doc };
  }

  async listAll(): Promise<UserDoc[]> {
    return Array.from(this.users.values()).map((u) => ({ ...u }));
  }

  async delete(id: string): Promise<void> {
    this.users.delete(id);
  }
}

import { IAppRepository } from '../db/repository.interface.js';
import {
  TodoDoc,
  CreateTodoInput,
  UpdateTodoInput,
  SortBy,
  SortOrder,
} from '../types/index.js';

export class TodoService {
  constructor(private repo: IAppRepository) {}

  async create(userId: string, input: CreateTodoInput): Promise<TodoDoc> {
    const todo: TodoDoc = {
      _id: crypto.randomUUID(),
      userId,
      title: input.title.trim(),
      description: input.description?.trim() || '',
      createdAt: new Date().toISOString(),
      dueDate: input.dueDate || null,
      completed: false,
      type: 'todo',
    };

    return await this.repo.todos.create(todo);
  }

  async list(
    userId: string,
    sortBy: SortBy = 'createdAt',
    order: SortOrder = 'desc',
  ): Promise<TodoDoc[]> {
    const todos = await this.repo.todos.listByUserId(userId);

    const isAsc = order === 'asc';

    return todos.sort((a, b) => {
      if (sortBy === 'title') {
        const cmp = a.title.localeCompare(b.title, undefined, {
          sensitivity: 'base',
        });
        return isAsc ? cmp : -cmp;
      }

      if (sortBy === 'dueDate') {
        const aHasDue = a.dueDate !== null && a.dueDate !== undefined && a.dueDate !== '';
        const bHasDue = b.dueDate !== null && b.dueDate !== undefined && b.dueDate !== '';

        // Todos without due date sort last regardless of order
        if (!aHasDue && !bHasDue) return 0;
        if (!aHasDue) return 1;
        if (!bHasDue) return -1;

        const dateA = new Date(a.dueDate!).getTime();
        const dateB = new Date(b.dueDate!).getTime();

        return isAsc ? dateA - dateB : dateB - dateA;
      }

      // Default: createdAt
      const dateA = new Date(a.createdAt).getTime();
      const dateB = new Date(b.createdAt).getTime();
      return isAsc ? dateA - dateB : dateB - dateA;
    });
  }

  async get(userId: string, id: string): Promise<TodoDoc | null> {
    const todo = await this.repo.todos.getById(id);
    if (!todo || todo.userId !== userId) {
      return null;
    }
    return todo;
  }

  async update(
    userId: string,
    id: string,
    input: UpdateTodoInput,
  ): Promise<TodoDoc> {
    const todo = await this.get(userId, id);
    if (!todo) {
      const err = new Error('Todo not found') as any;
      err.statusCode = 404;
      throw err;
    }

    if (input.title !== undefined) {
      todo.title = input.title.trim();
    }
    if (input.description !== undefined) {
      todo.description = input.description.trim();
    }
    if (input.dueDate !== undefined) {
      todo.dueDate = input.dueDate || null;
    }
    if (input.completed !== undefined) {
      todo.completed = input.completed;
    }

    return await this.repo.todos.update(todo);
  }

  async delete(userId: string, id: string): Promise<void> {
    const todo = await this.get(userId, id);
    if (!todo) {
      const err = new Error('Todo not found') as any;
      err.statusCode = 404;
      throw err;
    }

    await this.repo.todos.delete(id, todo._rev);
  }
}

import { generateToken, hashToken, getTokenDocId } from '../auth/crypto.js';

export class TokenService {
  constructor(private repo: IAppRepository) {}

  async createToken(
    userId: string,
    label: string,
  ): Promise<{ id: string; label: string; token: string; createdAt: string }> {
    const rawToken = `todo_mcp_${generateToken()}`;
    const tokenHash = hashToken(rawToken);
    const docId = getTokenDocId(rawToken);

    const doc: ApiTokenDoc = {
      _id: docId,
      userId,
      label: label.trim(),
      tokenHash,
      createdAt: new Date().toISOString(),
      revoked: false,
      type: 'api_token',
    };

    const saved = await this.repo.apiTokens.create(doc);

    return {
      id: saved._id,
      label: saved.label,
      token: rawToken,
      createdAt: saved.createdAt,
    };
  }

  async listTokens(
    userId: string,
  ): Promise<Array<{ id: string; label: string; createdAt: string; lastUsedAt?: string }>> {
    const tokens = await this.repo.apiTokens.listByUserId(userId);
    return tokens.map((t) => ({
      id: t._id,
      label: t.label,
      createdAt: t.createdAt,
      lastUsedAt: t.lastUsedAt,
    }));
  }

  async revokeToken(userId: string, id: string): Promise<void> {
    const token = await this.repo.apiTokens.getById(id);
    if (!token || token.userId !== userId) {
      const err = new Error('API token not found') as any;
      err.statusCode = 404;
      throw err;
    }

    token.revoked = true;
    await this.repo.apiTokens.update(token);
  }

  async validateToken(rawToken: string): Promise<AuthUser | null> {
    if (!rawToken) return null;
    const docId = getTokenDocId(rawToken);
    const tokenDoc = await this.repo.apiTokens.getById(docId);

    if (!tokenDoc || tokenDoc.revoked) {
      return null;
    }

    // Update lastUsedAt asynchronously
    tokenDoc.lastUsedAt = new Date().toISOString();
    this.repo.apiTokens.update(tokenDoc).catch(() => {});

    const user = await this.repo.users.getById(tokenDoc.userId);
    if (!user) return null;

    return {
      id: user._id,
      email: user.email,
      role: user.role,
      mustChangePassword: user.mustChangePassword,
    };
  }
}

import { IAppRepository } from '../db/repository.interface.js';
import { Config } from '../config.js';
import { UserDoc, AdminStatsResponse, UserStats } from '../types/index.js';
import {
  getUserDocId,
  normalizeEmail,
  generateAdminPassword,
} from '../auth/crypto.js';
import { hashPassword } from '../auth/password.js';

export class AdminService {
  constructor(private repo: IAppRepository, private config: Config) {}

  async bootstrapAdmin(): Promise<void> {
    const adminEmail = normalizeEmail(this.config.ADMIN_EMAIL);
    const adminId = getUserDocId(adminEmail);

    const existing = await this.repo.users.getById(adminId);
    if (existing) {
      return; // Admin already bootstrapped
    }

    const initialPassword = generateAdminPassword();
    const passwordHash = await hashPassword(initialPassword);

    const adminDoc: UserDoc = {
      _id: adminId,
      email: adminEmail,
      passwordHash,
      role: 'admin',
      mustChangePassword: true,
      createdAt: new Date().toISOString(),
      type: 'user',
    };

    try {
      await this.repo.users.create(adminDoc);
    } catch (err: any) {
      if (err.status === 409 || err.statusCode === 409) {
        return; // Another replica won the race
      }
      throw err;
    }

    // Deliver password via Kubernetes Secret or local secure file
    try {
      await this.deliverAdminPassword(adminEmail, initialPassword);
    } catch (err: any) {
      // Rollback admin doc on delivery failure
      try {
        await this.repo.users.delete(adminId);
      } catch {}
      throw new Error(`Failed to deliver initial admin password: ${err.message}`);
    }
  }

  private async deliverAdminPassword(
    adminEmail: string,
    password: string,
  ): Promise<void> {
    const k8sTokenPath = '/var/run/secrets/kubernetes.io/serviceaccount/token';
    let isKubernetes = false;
    try {
      await fs.access(k8sTokenPath);
      isKubernetes = true;
    } catch {}

    if (isKubernetes) {
      await this.createK8sSecret(adminEmail, password);
    } else {
      // Local dev: write to gitignored file with mode 0600
      const filePath = 'todo-admin-initial.txt';
      const content = `Admin initial credentials:\nEmail: ${adminEmail}\nPassword: ${password}\n\nPlease log in and change this password immediately.\n`;
      await fs.writeFile(filePath, content, { mode: 0o600 });
    }
  }

  private async createK8sSecret(
    adminEmail: string,
    password: string,
  ): Promise<void> {
    const token = await fs.readFile(
      '/var/run/secrets/kubernetes.io/serviceaccount/token',
      'utf-8',
    );
    const ca = await fs.readFile(
      '/var/run/secrets/kubernetes.io/serviceaccount/ca.crt',
    );

    const namespace = this.config.KUBERNETES_NAMESPACE;
    const secretName = this.config.ADMIN_SECRET_NAME;

    const payload = JSON.stringify({
      apiVersion: 'v1',
      kind: 'Secret',
      metadata: {
        name: secretName,
        namespace,
      },
      type: 'Opaque',
      stringData: {
        'admin-email': adminEmail,
        'admin-initial-password': password,
      },
    });

    return new Promise((resolve, reject) => {
      const req = https.request(
        {
          host: 'kubernetes.default.svc',
          port: 443,
          path: `/api/v1/namespaces/${namespace}/secrets`,
          method: 'POST',
          headers: {
            Authorization: `Bearer ${token}`,
            'Content-Type': 'application/json',
            'Content-Length': Buffer.byteLength(payload),
          },
          ca,
        },
        (res) => {
          if (res.statusCode && res.statusCode >= 200 && res.statusCode < 300) {
            resolve();
          } else if (res.statusCode === 409) {
            // Secret already exists
            resolve();
          } else {
            let data = '';
            res.on('data', (chunk) => (data += chunk));
            res.on('end', () =>
              reject(
                new Error(
                  `K8s Secret creation failed with HTTP ${res.statusCode}: ${data}`,
                ),
              ),
            );
          }
        },
      );

      req.on('error', reject);
      req.write(payload);
      req.end();
    });
  }

  async getStats(): Promise<AdminStatsResponse> {
    const allUsers = await this.repo.users.listAll();
    const regularUsers = allUsers.filter((u) => u.role !== 'admin');

    const allTodos = await this.repo.todos.listAll();
    const now = Date.now();

    const userStats: UserStats[] = regularUsers.map((user) => {
      const userTodos = allTodos.filter((t) => t.userId === user._id);
      const total = userTodos.length;
      const done = userTodos.filter((t) => t.completed).length;
      const open = total - done;
      const overdue = userTodos.filter(
        (t) =>
          !t.completed &&
          t.dueDate &&
          new Date(t.dueDate).getTime() < now,
      ).length;

      return {
        userId: user._id,
        email: user.email,
        total,
        done,
        open,
        overdue,
      };
    });

    return {
      totalUsers: regularUsers.length,
      users: userStats,
    };
  }

import { AuthUser } from '../types/index.js';
import { fetchJson } from '../api/client.js';

interface AuthContextType {
  user: AuthUser | null;
  loading: boolean;
  login: (email: string, pass: string) => Promise<AuthUser>;
  register: (email: string, pass: string) => Promise<AuthUser>;
  logout: () => Promise<void>;
  changePassword: (currentPass: string, newPass: string) => Promise<void>;
  refreshUser: () => Promise<void>;
}

const AuthContext = createContext<AuthContextType | undefined>(undefined);

export const AuthProvider: React.FC<{ children: React.ReactNode }> = ({
  children,
}) => {
  const [user, setUser] = useState<AuthUser | null>(null);
  const [loading, setLoading] = useState(true);

  const refreshUser = async () => {
    try {
      const res = await fetchJson<{ user: AuthUser }>('/api/auth/me');
      setUser(res.user);
    } catch {
      setUser(null);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    refreshUser();
  }, []);

  const login = async (email: string, pass: string): Promise<AuthUser> => {
    const res = await fetchJson<{ user: AuthUser }>('/api/auth/login', {
      method: 'POST',
      body: JSON.stringify({ email, password: pass }),
    });
    setUser(res.user);
    return res.user;
  };

  const register = async (email: string, pass: string): Promise<AuthUser> => {
    const res = await fetchJson<{ user: AuthUser }>('/api/auth/register', {
      method: 'POST',
      body: JSON.stringify({ email, password: pass }),
    });
    setUser(res.user);
    return res.user;
  };

  const logout = async () => {
    try {
      await fetchJson('/api/auth/logout', { method: 'POST' });
    } finally {
      setUser(null);
    }
  };

  const changePassword = async (currentPass: string, newPass: string) => {
    const res = await fetchJson<{ user: AuthUser }>('/api/auth/change-password', {
      method: 'POST',
      body: JSON.stringify({
        currentPassword: currentPass,
        newPassword: newPass,
      }),
    });
    setUser(res.user);
  };

  return (
    <AuthContext.Provider
      value={{
        user,
        loading,
        login,
        register,
        logout,
        changePassword,
        refreshUser,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
};

export function useAuth(): AuthContextType {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuth must be used within an AuthProvider');
  }
  return context;
}

  async delete(id: string, rev?: string): Promise<void> {
    let docRev = rev;
    if (!docRev) {
      const existing = await this.getById(id);
      if (!existing) return;
      docRev = existing._rev;
    }
    if (docRev) {
      await this.client.deleteDocument({
        db: this.db,
        docId: id,
        rev: docRev,
      });
    }
  }

  async listAll(): Promise<TodoDoc[]> {
    const res = await this.client.postAllDocs({
      db: this.db,
      includeDocs: true,
    });
    return (
      res.result.rows
        ?.filter((r) => r.doc && r.id && !r.id.startsWith('_design/'))
        .map((r) => r.doc as unknown as TodoDoc) || []
    );
  }
}

export class CloudantAppRepository implements IAppRepository {
  users: CloudantUserRepository;
  sessions: CloudantSessionRepository;
  apiTokens: CloudantApiTokenRepository;
  todos: CloudantTodoRepository;

  constructor(private client: CloudantV1, private config: Config) {
    this.users = new CloudantUserRepository(client, config.DB_USERS);
    this.sessions = new CloudantSessionRepository(client, config.DB_SESSIONS);
    this.apiTokens = new CloudantApiTokenRepository(client, config.DB_API_TOKENS);
    this.todos = new CloudantTodoRepository(client, config.DB_TODOS);
  }

  async init(): Promise<void> {
    const dbs = [
      this.config.DB_USERS,
      this.config.DB_SESSIONS,
      this.config.DB_API_TOKENS,
      this.config.DB_TODOS,
    ];

    for (const db of dbs) {
      try {
        await this.client.putDatabase({ db });
      } catch (err: any) {
        if (err.status !== 412 && err.statusCode !== 412) {
          // 412 Precondition Failed = database already exists
          // If we lack permissions to create, verify access
          try {
            await this.client.getDatabaseInformation({ db });
          } catch (innerErr) {
            throw new Error(`Failed to access or create Cloudant database "${db}": ${err.message}`);
          }
        }
      }
    }

    // Create Mango index on userId for todos
    try {
      await this.client.postIndex({
        db: this.config.DB_TODOS,
        index: {
          fields: [{ userId: 'asc' }],
        },
        name: 'userId-index',
        type: 'json',
      });
    } catch {
      // Index creation might already exist or lack permission, which is acceptable
    }

    // Create Mango index on userId for api_tokens
    try {
      await this.client.postIndex({
        db: this.config.DB_API_TOKENS,
        index: {
          fields: [{ userId: 'asc' }, { revoked: 'asc' }],
        },
        name: 'userId-revoked-index',
        type: 'json',
      });
    } catch {
      // Ignored if already exists
    }
  }
}

export function createCloudantClient(cfg: Config): CloudantV1 {
  let authenticator;
  if (cfg.CLOUDANT_APIKEY) {
    authenticator = new IamAuthenticator({
      apikey: cfg.CLOUDANT_APIKEY,
    });
  } else {
    authenticator = new BasicAuthenticator({
      username: cfg.CLOUDANT_USERNAME || 'admin',
      password: cfg.CLOUDANT_PASSWORD || '',
    });
  }

  return CloudantV1.newInstance({
    authenticator,
    serviceUrl: cfg.CLOUDANT_URL,
  });
}
